# Приложение — единственная точка композиции независимых модулей.
# Исходник app.cpp не обязан использовать какой-либо модуль: приложение
# должно успешно собираться даже с тремя выключенными опциями.
add_executable(app app.cpp)

# Связываем только реально объявленные цели. Здесь допустимо подключать
# все модули вместе: это не создаёт зависимостей МЕЖДУ библиотеками.
# PRIVATE означает, что зависимости нужны приложению, а не его потребителям.
foreach(_module IN ITEMS pgr_core pgr_gui pgr_nn)
    if(TARGET "${_module}")
        target_link_libraries(app PRIVATE "${_module}")
    endif()
endforeach()

# Стандарт C++17 задан в корневом файле до создания цели app.
